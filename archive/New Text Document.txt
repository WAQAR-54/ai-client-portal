    const target=parseFloat(el.dataset.target);
    const suffix=el.dataset.suffix||'';
    const isDecimal=String(el.dataset.target).includes('.');
    const duration=800;
    const start=performance.now();
    function frame(now){
      const p=Math.min((now-start)/duration,1);
      const eased=1-Math.pow(1-p,3);
      const val=target*eased;
      el.textContent=(isDecimal?val.toFixed(1):Math.round(val))+suffix;
      if(p<1) requestAnimationFrame(frame);
    }
    requestAnimationFrame(frame);
  }

    demo:{models:['haiku'],requests:10,period:'day',context:4000,expiry:2,features:{upload:false,tools:false,priority:false}},
    basic:{models:['sonnet'],requests:50,period:'day',context:16000,expiry:'',features:{upload:true,tools:false,priority:false}},
    advanced:{models:['opus','gpt'],requests:200,period:'day',context:64000,expiry:'',features:{upload:true,tools:true,priority:false}},
    full:{models:['haiku','sonnet','opus','fable','gpt'],requests:1000,period:'day',context:128000,expiry:'',features:{upload:true,tools:true,priority:true}}
  };
  let editingPlan=null;
  function openEditPlanModal(planId){
    editingPlan=planId;
    const d=planData[planId];
    document.getElementById('editPlanTitle').textContent='Edit '+planId.charAt(0).toUpperCase()+planId.slice(1)+' plan';
    ['haiku','sonnet','opus','fable','gpt'].forEach(m=>{ document.getElementById('chk-'+m).checked = d.models.includes(m); });
    document.getElementById('editRequests').value=d.requests;
    document.getElementById('editPeriod').value=d.period;
    document.getElementById('editContext').value=d.context;
    document.getElementById('editExpiry').value=d.expiry;
    document.getElementById('feat-upload').classList.toggle('on', d.features.upload);
    document.getElementById('feat-tools').classList.toggle('on', d.features.tools);
    document.getElementById('feat-priority').classList.toggle('on', d.features.priority);
    openModal('editPlanModal');
  }
  function submitEditPlan(){
    if(!editingPlan) return;
    const p=editingPlan;
    const requests=document.getElementById('editRequests').value;
    const period=document.getElementById('editPeriod').value;
    const context=document.getElementById('editContext').value;
    const expiry=document.getElementById('editExpiry').value;
    const upload=document.getElementById('feat-upload').classList.contains('on');
    const tools=document.getElementById('feat-tools').classList.contains('on');
    const priority=document.getElementById('feat-priority').classList.contains('on');

    document.getElementById(p+'-requests').textContent=`${requests} / ${period}`;
    document.getElementById(p+'-context').textContent=`${Math.round(context/1000)}k tokens`;
    document.getElementById(p+'-expiry').textContent=expiry ? `${expiry} days` : 'No expiry';

    const names={haiku:'Haiku',sonnet:'Sonnet',opus:'Opus',fable:'Fable',gpt:'GPT-5.6 Terra'};
    const selected=['haiku','sonnet','opus','fable','gpt'].filter(m=>document.getElementById('chk-'+m).checked).map(m=>names[m]);
    document.getElementById(p+'-models').textContent = selected.length ? selected.join(', ') : 'None selected';

    const featEl=document.getElementById(p+'-feats');
    featEl.innerHTML =
      `<span class="feat-tag ${upload?'on':'off'}">File upload</span>` +
      `<span class="feat-tag ${tools?'on':'off'}">Tools</span>` +
      (priority ? '<span class="feat-tag on">Priority</span>' : '');

    const card=document.getElementById('plancard-'+p);
    card.style.animation='none'; card.offsetHeight; card.style.animation='rise .4s var(--ease) both';
    closeModal('editPlanModal');
  }
